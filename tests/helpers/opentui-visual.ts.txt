// Renders the generated OpenTUI project at 100x40 (OpenTUI's test renderer under Bun) and prints
// the character frame between FRAME-BEGIN / FRAME-END. Copied next to the generated project as visual.ts.
import { createTestRenderer } from '@opentui/core/testing';
import { Ui } from './src/index';

const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
const setup = await createTestRenderer({ width: 100, height: 40, exitOnCtrlC: false });
const ui = new Ui();
await ui.start({ renderer: setup.renderer });
await sleep(200);
await setup.renderOnce();
console.log('FRAME-BEGIN');
console.log(setup.captureCharFrame());
console.log('FRAME-END');
ui.stop();
process.exit(0);
